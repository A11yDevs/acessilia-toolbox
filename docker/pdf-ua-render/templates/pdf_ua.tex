% Alternate text of the next formula, for its Formula structure element.
% latex-lab builds /Alt from \l__math_content_template_tl ("LaTeX formula
% starts ... ends"); filters/formula_alt.lua wraps a formula in a group that
% calls this command, so the override ends with the group and the following
% formulas keep the default. The argument comes with TeX specials removed.
\ExplSyntaxOn
\NewDocumentCommand \AccessiliaFormulaAlt { m }
  { \tl_set:Nn \l__math_content_template_tl { \exp_not:n {#1} } }

% Turns math collection back on for the rest of the group, so the next formula
% gets its Formula element: table alignment (latex-lab-testphase-table) can
% switch collection off while it builds a cell. \MathCollectTrue is internal to
% latex-lab (latex-lab-math.ltx; despite the name it sets
% \l__math_collected_bool to false, the flag that suppresses collection), not
% a documented interface. The image pins TeX Live 2025 and its build smoke
% renders a formula in a table cell; if a latex-lab update drops the command,
% this check names it instead of failing on an undefined control sequence.
\msg_new:nnn { accessilia } { math-collect-missing }
  { latex-lab~no~longer~defines~\iow_char:N\\MathCollectTrue;~formulas~in~table~cells~would~lose~their~Formula~element. }
\AtBeginDocument
  {
    \cs_if_exist:NF \MathCollectTrue
      { \msg_error:nn { accessilia } { math-collect-missing } }
  }
\NewDocumentCommand \AccessiliaFormulaCollect { } { \MathCollectTrue }
\ExplSyntaxOff

% unicode-math sets a math letter as its Mathematical Alphanumeric Symbol
% (v is U+1D463 MATHEMATICAL ITALIC SMALL V), and the font's ToUnicode CMap
% gives that plane-1 character to the text layer. A reader that ignores the
% Formula /Alt (a browser viewer, read aloud, search, copy and paste) then gets
% "símbolo 1a" for a variable. Each math font loaded from here on maps its
% italic and bold italic Latin and Greek letters (and the italic h, dotless i
% and j) to their base letters; the glyphs stay the same. Double-struck,
% script, fraktur, sans and monospace letters keep their code points, which
% carry meaning (R is not ℝ). A math font is one with the math alphabet
% (U+1D44E): the text fonts are left alone, so an ℎ (U+210E, which DejaVu Sans
% has) written in prose keeps its code point. It must come before the first
% font is loaded.
% The Lua below has no TeX specials (percent, hash, tilde, backslash) and no
% comments: \directlua reads it as one line.
\directlua{
  local function hex4(code)
    local digits = "0123456789ABCDEF"
    local out = ""
    for _ = 1, 4 do
      local digit = code - 16 * math.floor(code / 16)
      out = digits:sub(digit + 1, digit + 1) .. out
      code = math.floor(code / 16)
    end
    return out
  end
  local base = {}
  for _, block in ipairs({{0x1D434, 0x41}, {0x1D44E, 0x61}, {0x1D468, 0x41}, {0x1D482, 0x61}}) do
    for offset = 0, 25 do base[block[1] + offset] = block[2] + offset end
  end
  base[0x210E] = 0x68
  base[0x1D6A4] = 0x131
  base[0x1D6A5] = 0x237
  local greek = {}
  for offset = 0, 24 do greek[offset] = 0x391 + offset end
  greek[17] = 0x3F4
  greek[25] = 0x2207
  for offset = 0, 24 do greek[26 + offset] = 0x3B1 + offset end
  for offset, code in ipairs({0x2202, 0x3F5, 0x3D1, 0x3F0, 0x3D5, 0x3F1, 0x3D6}) do
    greek[50 + offset] = code
  end
  for _, first in ipairs({0x1D6E2, 0x1D71C}) do
    for offset = 0, 57 do base[first + offset] = greek[offset] end
  end
  luatexbase.add_to_callback("luaotfload.patch_font", function(tfmdata)
    local characters = tfmdata.characters
    if not characters or not characters[0x1D44E] then return end
    for code, letter in pairs(base) do
      local character = characters[code]
      if character then character.tounicode = hex4(letter) end
    end
  end, "accessilia.math_letter_tounicode")
}

% unicode-math sets formulas in TeX Gyre DejaVu Math, the OpenType math font
% matching the DejaVu Sans text (the default, Latin Modern, is serif), and gives
% each symbol its own code point: without it \neq is \not overlaid on =, and

% with amssymb, so amssymb is not loaded.
