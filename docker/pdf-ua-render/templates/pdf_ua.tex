% Math from the core is LaTeX written by recognizers and vision models, which
% use amsmath environments (aligned). Without them one formula fails the render
% and the core degrades every formula of the document to literal text. The
% symbols (\mathbb, \neq, ...) come from unicode-math, loaded after fontspec.
\usepackage{amsmath}

% Alternate text of the next formula, for its Formula structure element.
% latex-lab builds /Alt from \l__math_content_template_tl ("LaTeX formula
% starts ... ends"); filters/formula_alt.lua wraps a formula in a group that
% calls this command, so the override ends with the group and the following
% formulas keep the default. The argument comes with TeX specials removed.
\ExplSyntaxOn
\NewDocumentCommand \AccessiliaFormulaAlt { m }
  { \tl_set:Nn \l__math_content_template_tl { \exp_not:n {#1} } }

% Turns math collection back on for the rest of the group, so the next formula
% gets its Formula element: table alignment (latex-lab-testphase-table) can
% switch collection off while it builds a cell. \MathCollectTrue is internal to
% latex-lab (latex-lab-math.ltx; despite the name it sets
% \l__math_collected_bool to false, the flag that suppresses collection), not
% a documented interface. The image pins TeX Live 2025 and its build smoke
% renders a formula in a table cell; if a latex-lab update drops the command,
% this check names it instead of failing on an undefined control sequence.
\msg_new:nnn { accessilia } { math-collect-missing }
  { latex-lab~no~longer~defines~\iow_char:N\\MathCollectTrue;~formulas~in~table~cells~would~lose~their~Formula~element. }
\AtBeginDocument
  {
    \cs_if_exist:NF \MathCollectTrue
      { \msg_error:nn { accessilia } { math-collect-missing } }
  }
\NewDocumentCommand \AccessiliaFormulaCollect { } { \MathCollectTrue }
\ExplSyntaxOff

% unicode-math sets formulas in TeX Gyre DejaVu Math, the OpenType math font
% matching the DejaVu Sans text (the default, Latin Modern, is serif), and gives
% each symbol its own code point: without it \neq is \not overlaid on =, and
% the text layer reads "=" plus an orphan U+0338 instead of U+2260. It conflicts
% with amssymb, so amssymb is not loaded.
\usepackage{unicode-math}
\setmathfont{texgyredejavu-math.otf}
% Symbols amssymb draws that TeX Gyre DejaVu Math has no glyph for (\bigstar,
% \subseteqq, \pitchfork, ...): without a fallback they render as nothing.

  "25B4,"25B5,"25BE,"25BF,"2605,"27CB,"27CD,"2A5E,"2AB5-"2ABA,"2AC5,"2AC6,
  "29EB,"2ACB,"2ACC}]
